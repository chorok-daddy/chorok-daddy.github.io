import RPi.GPIO as GPIO
import time
from threading import Timer

def infinite_LED_blink():
    GPIO.output(5, True)
    time.sleep(0.1)
    GPIO.output(5, False)
    Timer(2, infinite_LED_blink).start()
    
try:
    GPIO.setmode(GPIO.BCM)
    GPIO.setup(5, GPIO.OUT)
    infinite_LED_blink()
    while True:
        pass
finally:
    print('cleaning up')
    GPIO.cleanup()
