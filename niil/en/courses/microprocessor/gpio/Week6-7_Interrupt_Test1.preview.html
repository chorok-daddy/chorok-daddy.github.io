import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)
GPIO.setup(6, GPIO.IN)

try:
    while True:
        GPIO.wait_for_edge(6, GPIO.BOTH)
        if not GPIO.input(6):
            print('Button Pressed!')
        else:
            print('Button Released!')
finally:
    print('cleaning up')
    GPIO.cleanup()
