# Design the Model
class Logistic(nn.Module):
    def __init__(self, num_inputs, num_outputs):
        super().__init__()
        self.linear = nn.Linear(num_inputs, num_outputs)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        return self.sigmoid(self.linear(x))

# Initialize the Model (input dim, output dim)
model = Logistic(2, 1)

2 Hypothesis, Cost and Optimization

  • Assume the following equations hold.
    • For 1 input (scalar): yhypo=H(x)=11+e−αxy_{hypo} = H(x) = \frac{1}{1+e^{-\alpha x}}
    • For multiple inputs (vector): yhypo=H(X)=11+e−WTXy_{hypo} = H(X) = \frac{1}{1+e^{-W^TX}}
  • For example, with 3 data points: y=11+e−[w1w2w3]⋅[x1x2x3]y = \frac{1}{1 + e^{-\begin{bmatrix} w_1 && w_2 && w_3 \end{bmatrix} \cdot \begin{bmatrix} x_1 \\ x_2 \\ x_3 \end{bmatrix}}}
    • Find WW that satisfies this.
  • Use binary cross entropy as the cost function. −[ytrainlog(yhypo)+(1−ytrain)log(1−yhypo)]-[y_{train}log(y_{hypo}) + (1-y_{train})log(1-y_{hypo})]
# Configure the Optimizer (set learning rate = 1)
optimizer = opt.SGD(model.parameters(), lr=1)

# Repeat
for epoch in range(100):

  # Compute Cost / mse_loss(predicted value, true value)
  y_hypo = model(x_train)
  cost = func.binary_cross_entropy(y_hypo, y_train)

  # Update the model using cost
  optimizer.zero_grad()
  cost.backward()
  optimizer.step()

  # Print intermediate results every 10 iterations
  if epoch % 10 == 9:
    prediction = y_hypo >= torch.FloatTensor([0.5])
    correct_prediction = prediction.float() == y_train
    accuracy = correct_prediction.sum().item() / len(correct_prediction)
    print('Epoch {:4d}/{} Cost: {:.6f} Accuracy {:2.2f}%'.format(
            epoch, 100, cost.item(), accuracy * 100,
    ))
