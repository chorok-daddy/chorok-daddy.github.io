Run the sample code (Random CartPole)

  • Simulate Cartpole with random control inputs
  • Without effective control, the pole falls and the episode ends after only 1–2 steps.
def run_random_agent(env_name="CartPole-v1", max_episodes=2, max_steps=200):
    env = gym.make(env_name)
    for episode in range(max_episodes):
        state, info = env.reset()
        total_reward = 0
        for step in range(max_steps):
            action = env.action_space.sample()  # Random action
            next_state, reward, terminated, truncated, info = env.step(action)
            total_reward += reward
            state = next_state
            if terminated or truncated:
                print(f"Episode {episode+1} finished after {step+1} steps, total reward: {total_reward}")
                break
    env.close()
