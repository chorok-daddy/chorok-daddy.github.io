wn=5;zeta_list=[.1 .3 .7 1.2];
figure;hold on;
for zeta=zeta_list
T=tf(wn^2,[1 2*zeta*wn wn^2]);[y,t]=step(T,0:.005:20);plot(t,y,'DisplayName',sprintf('zeta=%.1f',zeta));
info=stepinfo(T);disp([zeta info.Overshoot info.SettlingTime]);
